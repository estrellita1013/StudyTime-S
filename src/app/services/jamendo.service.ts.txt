import { Injectable } from '@angular/core';
import { HttpClient, HttpParams } from '@angular/common/http';
import { firstValueFrom } from 'rxjs';

export interface JamendoTrack {
  id: string;
  name: string;
  artistName: string;
  albumName: string;
  duration: number;
  audio: string;
  image: string;
  shareUrl: string;
  licenseUrl: string;
}

interface JamendoResponse {
  headers: {
    status: string;
    code: number;
    error_message: string;
    results_count: number;
  };

  results: Array<{
    id: string;
    name: string;
    artist_name: string;
    album_name?: string;
    duration: number;
    audio: string;
    image?: string;
    album_image?: string;
    shareurl?: string;
    license_ccurl?: string;
  }>;
}

@Injectable({
  providedIn: 'root'
})
export class JamendoService {

  private readonly apiUrl =
    'https://api.jamendo.com/v3.0/tracks/';

  /*
   * ID de prueba proporcionado por Jamendo.
   *
   * IMPORTANTE:
   * Este ID es solamente para pruebas del API de lectura.
   *
   * Cuando tengas tu propia aplicación en Jamendo,
   * sustituye este valor por tu client_id.
   */
  private readonly clientId = '709fa152';

  constructor(
    private http: HttpClient
  ) {}

  async buscarMusica(
    texto: string,
    limite: number = 10
  ): Promise<JamendoTrack[]> {

    const termino = texto.trim();

    if (!termino) {
      return [];
    }

    const params = new HttpParams()
      .set('client_id', this.clientId)
      .set('format', 'json')
      .set('search', termino)
      .set('limit', limite.toString())
      .set('audioformat', 'mp31')
      .set('imagesize', '300')
      .set('type', 'single albumtrack');

    try {

      const respuesta = await firstValueFrom(
        this.http.get<JamendoResponse>(
          this.apiUrl,
          { params }
        )
      );

      if (
        !respuesta ||
        !respuesta.results
      ) {
        return [];
      }

      return respuesta.results
        .filter(track => !!track.audio)
        .map(track => ({
          id: track.id,
          name: this.limpiarTexto(track.name),
          artistName: this.limpiarTexto(track.artist_name),
          albumName: this.limpiarTexto(
            track.album_name || 'Sin álbum'
          ),
          duration: track.duration || 0,
          audio: track.audio,
          image:
            track.image ||
            track.album_image ||
            '',
          shareUrl:
            track.shareurl ||
            `https://www.jamendo.com/track/${track.id}`,
          licenseUrl:
            track.license_ccurl || ''
        }));

    } catch (error: any) {

      console.error(
        '========== ERROR JAMENDO =========='
      );

      console.error(
        'Status:',
        error?.status
      );

      console.error(
        'Mensaje:',
        error?.message
      );

      console.error(
        'Respuesta:',
        error?.error
      );

      console.error(
        '==================================='
      );

      throw error;
    }
  }

  private limpiarTexto(
    texto: string
  ): string {

    return texto
      .replace(/&amp;/g, '&')
      .replace(/&quot;/g, '"')
      .replace(/&#39;/g, "'")
      .replace(/&lt;/g, '<')
      .replace(/&gt;/g, '>');
  }
}
